4. **Metrics Middleware**
   ```go
   func MetricsMiddleware(next http.Handler) http.Handler {
       return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
           start := time.Now()
           
           // Wrap response writer to capture status
           ww := &responseWriter{ResponseWriter: w}
           next.ServeHTTP(ww, r)
           
           // Record metrics
           duration := time.Since(start).Milliseconds()
           ProviderLatencyMs.WithLabelValues(r.URL.Path).Observe(float64(duration))
           RequestsTotal.WithLabelValues(r.URL.Path, strconv.Itoa(ww.status)).Inc()
       })
   }
   ```

6. **Provider Health Monitor**
   ```go
   // Background goroutine: check provider health periodically
   func (m *ProviderMonitor) Start(interval time.Duration) {
       ticker := time.NewTicker(interval)
       for range ticker.C {
           for _, conn := range m.providers.List() {
               err := m.testConnection(conn)
               if err != nil {
                   m.log.Warn("provider health check failed",
                       "provider", conn.Provider,
                       "connection", conn.Name,
                       "error", err,
                   )
               }
           }
       }
   }
   ```

## Pseudocode

```
GET /health:
  result := {
    ok: true,
    version: Version,
    uptime: time.Since(startTime).Seconds(),
  }
  
  // Check DB
  if err := db.Ping(); err != nil:
    result.db = "error: " + err.Error()
    result.ok = false
  else:
    result.db = "ok"
  
  // Check Redis (if enabled)
  if config.RedisURL != "":
    if err := redis.Ping(); err != nil:
      result.redis = "error: " + err.Error()
    else:
      result.redis = "ok"
  else:
    result.redis = "disabled"
  
  // Check providers
  result.providers = {}
  for provider in providers.List():
    active := providers.CountActive(provider)
    cooldown := providers.CountCooldown(provider)
    result.providers[provider] = {
      status: "ok" if cooldown == 0 else "degraded",
      active_accounts: active,
      cooldown_accounts: cooldown,
    }
  
  if !result.ok:
    w.WriteHeader(503)
  json(w, result)
```
