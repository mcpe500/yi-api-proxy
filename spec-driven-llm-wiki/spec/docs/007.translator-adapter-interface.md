# Spec 007: Translator/Adapter Interface (Normalized Types, ProviderAdapter)

## Task/Prompt

Implementasi translator/adapter interface:
1. Internal normalized request/response types
2. ProviderAdapter interface
3. Adapter registry
4. Request translation: OpenAI format → provider format
5. Response translation: provider format → OpenAI format
6. Stream event normalization
7. Error normalization

Referensi 9Router: translator registry, format translation (OpenAI/Claude/Gemini/CommandCode).

## Tujuan

Arsitektur adapter yang:
- Single interface untuk semua provider
- Internal normalized format (OpenAI-compatible)
- Setiap provider hanya implementasi adapter
- Mudah tambah provider baru

## Kenapa Tujuan Ini

Adapter interface adalah foundation spec 008 (OpenAI, OpenAI-compatible), spec 009 (GLM, MiniMax), spec 015 (CommandCode, OpenCode Go). Tanpa interface ini, tidak ada konsistensi antara provider adapters.

## Logic Changes

1. **Normalized Chat Request**
   ```go
   type NormalizedChatRequest struct {
       RequestID       string
       UserID          string
       Model           string  // "openai/gpt-4o"
       Messages        []Message
       Tools           []Tool
       ToolChoice      any     // "auto", "none", or ToolCall object
       Stream          bool
       Temperature      *float64
       TopP             *float64
       MaxTokens        *int
       ResponseFormat   *ResponseFormat
       Stop             []string
       PresencePenalty   *float64
       FrequencyPenalty  *float64
       User             string
       Metadata         map[string]string
   }

   type Message struct {
       Role    string  // "system", "user", "assistant", "tool"
       Content any     // string or []ContentPart
       Name    string
       ToolCallID string
   }

   type Tool struct {
       Type    string  // "function"
       Function Function
   }

   type Function struct {
       Name        string
       Description string
       Parameters  any  // JSON Schema
   }

   type ResponseFormat struct {
       Type string  // "text", "json_object"
   }
   ```

2. **Normalized Chat Response**
   ```go
   type NormalizedChatResponse struct {
       ID      string
       Model   string
       Choices []Choice
       Usage   Usage
       Created int64  // Unix timestamp
       Raw     []byte // Original provider response for debugging
   }

   type Choice struct {
       Index        int
       Message      Message
       FinishReason string  // "stop", "length", "content_filter"
       Delta        Message // For streaming
   }

   type Usage struct {
       PromptTokens     int
       CompletionTokens int
       TotalTokens      int
   }
   ```

3. **Normalized Stream Event**
   ```go
   type StreamEvent struct {
       ID      string
       Model   string
       Delta   Message // Partial content
       Usage   *Usage  // Optional, for final chunk
       Done    bool    // True for [DONE]
   }
   ```

4. **ProviderAdapter Interface**
   ```go
   type ProviderAdapter interface {
       // Metadata
       Name() string
       Provider() string
       Capabilities() Capabilities
       
       // Request handling
       TranslateRequest(ctx context.Context, req NormalizedChatRequest, cred *Credential) (*http.Request, error)
       ParseResponse(ctx context.Context, resp *http.Response, req NormalizedChatRequest) (*NormalizedChatResponse, error)
       StreamResponse(ctx context.Context, resp *http.Response, req NormalizedChatRequest) (<-chan StreamEvent, error)
       
       // Credential handling
       RefreshCredentials(ctx context.Context, cred *Credential) (*Credential, error)
       TestConnection(ctx context.Context, cred *Credential) error
       
       // Model info
       ListModels(ctx context.Context, cred *Credential) ([]Model, error)
   }

   type Capabilities struct {
       SupportsChat        bool
       SupportsStreaming   bool
       SupportsTools        bool
       SupportsVision       bool
       SupportsJSONMode     bool
       SupportsEmbeddings   bool
   }
   ```

5. **Adapter Registry**
   ```go
   type AdapterRegistry struct {
       adapters map[string]ProviderAdapter
       mu       sync.RWMutex
   }

   func (r *AdapterRegistry) Register(adapter ProviderAdapter)
   func (r *AdapterRegistry) Get(provider string) ProviderAdapter
   func (r *AdapterRegistry) List() []ProviderAdapter
   ```

6. **Request Translation Flow**
   ```
   TranslateToProvider(provider, req):
     adapter = registry.Get(provider)
     cred = decryptCredential(providerConn.EncryptedSecret)
     upstreamReq, err = adapter.TranslateRequest(ctx, req, cred)
     return upstreamReq
   ```

7. **Response Translation Flow**
   ```
   TranslateFromProvider(provider, upstreamResp, req):
     adapter = registry.Get(provider)
     normalized, err = adapter.ParseResponse(ctx, upstreamResp, req)
     return normalized  // OpenAI-compatible format
   ```

8. **Stream Translation Flow**
   ```
   TranslateStream(provider, upstreamStream, req):
     adapter = registry.Get(provider)
     events = adapter.StreamResponse(ctx, upstreamResp, req)
     
     // Stream events one by one
     for event := range events:
       yield event  // Already normalized to OpenAI SSE format
   ```

9. **Error Normalization**
   ```
   NormalizeError(provider, statusCode, body):
     // Parse provider-specific error format
     // Map to OpenAI-compatible error
     return OpenAIError{
       Message: humanReadableMessage,
       Type:    errorType,      // "invalid_request_error", "rate_limit_error", etc.
       Code:    errorCode,      // "rate_limit_exceeded", "model_not_found", etc.
     }
   ```

## Pseudocode

```
// Request translation example: OpenAI → GLM
type GLMAdapter struct {
  baseURL string
}

func (a *GLMAdapter) TranslateRequest(ctx, req, cred):
  // GLM uses OpenAI-compatible format mostly, but has differences
  upstream := mapOpenAIRequestToGLM(req)
  
  body := json.Marshal(upstream)
  httpReq, _ := http.NewRequestWithContext(ctx, "POST", a.baseURL+"/v1/chat/completions", bytes.NewReader(body))
  httpReq.Header.Set("Authorization", "Bearer "+cred.APIKey)
  httpReq.Header.Set("Content-Type", "application/json")
  
  return httpReq, nil

func (a *GLMAdapter) ParseResponse(ctx, resp, req):
  body, _ := io.ReadAll(resp.Body)
  
  // GLM response is mostly OpenAI-compatible, just map finish_reason
  glmResp := parseJSON(body)
  normalized := NormalizedChatResponse{
    ID:      glmResp.id,
    Model:   glmResp.model,
    Choices: glmResp.choices.Map(c => ({
      Index:        c.index,
      Message:      c.message,
      FinishReason: mapGLMFinishReason(c.finish_reason),
    })),
    Usage: glmResp.usage,
    Created: glmResp.created,
  }
  return normalized, nil

// Error normalization
NormalizeError(provider, status, body):
  switch provider:
    "openai":
      return parseOpenAIError(body)
    "glm":
      return parseGLMError(body)
    "minimax":
      return parseMiniMaxError(body)
  
  // Fallback: map by status code
  return MapStatusCodeToError(status)
```
